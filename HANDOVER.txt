// rust
// ============================================================================
// v197 — Police-escrow PGP sealing (user request): a second auto-download
// whose payload is a real OpenPGP message (PKESK v3 + SEIPD v1) readable only
// by the recipient's PRIVATE key, plus the attested hashes in CLEAR. Crypto
// runs HERE (server-side) so the callback tab stays ciphertext-only. The
// algorithm set is byte-for-byte the stack proven against gpg 2.4.9 in the
// zk-rsa.html four-leg interop (20b1ceb): no CFB resync (§5.13), MDC SHA-1,
// PKCS#1 v1.5 session-key wrap. No new dependencies (openssl/rand/sha2/hex
// are all present).
// ============================================================================
mod pgp_escrow {
    use sha2::Digest;
 
    const PGP_SYM_AES256: u8 = 9; // RFC 4880 §9.2
 
    struct Packet { tag: u8, body: Vec<u8> }
    struct Target { fingerprint: String, keyid: String, n: Vec<u8>, e: Vec<u8> }
 
    fn crc24(data: &[u8]) -> u32 {
        let mut crc: u32 = 0xb704ce;
        for &b in data {
            crc ^= (b as u32) << 16;
            for _ in 0..8 {
                crc <<= 1;
                if crc & 0x100_0000 != 0 { crc ^= 0x1864cfb; }
            }
        }
        crc & 0xff_ffff
    }
 
    // Armored block → raw bytes (regex-free: find BEGIN/END markers, first
    // blank line starts base64, an "=XXXX" line is the optional CRC —
    // mirroring zkRsaPgpPackets' b64 handling; CRC-verified when present).
    fn dearmor(armor: &str) -> Result<Vec<u8>, String> {
        let begin = "-----BEGIN PGP PUBLIC KEY BLOCK-----";
        let end = "-----END PGP PUBLIC KEY BLOCK-----";
        let s = armor.trim();
        let b = s.find(begin).ok_or("not a PGP PUBLIC KEY BLOCK")? + begin.len();
        let e = s[b..].find(end).ok_or("missing armor end")? + b;
        let body = &s[b..e];
        let body = body.split_once("\n\n").map(|(_, r)| r).ok_or("missing armor blank line")?;
        let (data, crc_line) = match body.rsplit_once("\n=") {
            Some((d, c)) => (d, Some(c.lines().next().unwrap_or(""))),
            None => (body, None),
        };
        let clean: String = data.chars().filter(|c| !c.is_whitespace()).collect();
        let bin = base64::engine::general_purpose::STANDARD
            .decode(clean.trim_end_matches('='))
            .map_err(|e| format!("armor base64: {e}"))?;
        if let Some(c) = crc_line {
            let want = base64::engine::general_purpose::STANDARD
                .decode(c.trim().trim_start_matches('='))?;
            if want.len() != 3 || crc24(&bin)
                != ((want[0] as u32) << 16 | (want[1] as u32) << 8 | want[2] as u32) {
                return Err("armor CRC mismatch — truncated or tampered".into());
            }
        }
        Ok(bin)
    }
 
    fn parse_packets(bin: &[u8]) -> Result<Vec<Packet>, String> {
        let mut out = Vec::new();
        let mut i = 0usize;
        while i < bin.len() {
            let ctb = bin[i]; i += 1;
            if ctb & 0x80 == 0 { return Err("malformed packet header".into()); }
            let (tag, len) = if ctb & 0x40 != 0 {
                let tag = ctb & 0x3f;
                let l = if i >= bin.len() { return Err("truncated".into()); }
                    else if bin[i] < 192 { let l = bin[i] as usize; i += 1; l }
                    else if bin[i] < 224 {
                        if i + 1 >= bin.len() { return Err("truncated".into()); }
                        let l = (((bin[i] as usize) - 192) << 8) + bin[i + 1] as usize + 192;
                        i += 2; l
                    } else if bin[i] == 255 {
                        if i + 4 >= bin.len() { return Err("truncated".into()); }
                        let l = u32::from_be_bytes([bin[i+1], bin[i+2], bin[i+3], bin[i+4]]) as usize;
                        i += 5; l
                    } else { return Err("partial-body lengths unsupported".into()); };
                (tag, l)
            } else {
                let tag = (ctb >> 2) & 0x0f;
                let len = match ctb & 0x03 {
                    0 => { let l = bin.get(i).copied().ok_or("truncated")? as usize; i += 1; l }
                    1 => { if i + 1 >= bin.len() { return Err("truncated".into()); }
                           let l = u16::from_be_bytes([bin[i], bin[i+1]]) as usize; i += 2; l }
                    2 => { if i + 3 >= bin.len() { return Err("truncated".into()); }
                           let l = u32::from_be_bytes([bin[i], bin[i+1], bin[i+2], bin[i+3]]) as usize; i += 4; l }
                    _ => bin.len() - i, // indeterminate
                };
                (tag, len)
            };
            if i + len > bin.len() { return Err("packet overruns buffer".into()); }
            out.push(Packet { tag, body: bin[i..i + len].to_vec() });
            i += len;
        }
        Ok(out)
    }
 
    // v4 public-key material: created(4) algo(1) [RSA: n MPI, e MPI] →
    // (material-without-mpis, n, e). Rejects non-RSA primaries exactly like
    // the JS parser (zkRsaParsePgpKeyPacket) — the Bayern key is RSA.
    fn rsa_key(body: &[u8]) -> Result<(Vec<u8>, Vec<u8>, Vec<u8>), String> {
        if body.len() < 6 || body[0] != 4 { return Err("not a v4 key packet".into()); }
        let algo = body[5];
        if algo != 1 && algo != 2 && 3 != algo && algo != 3 { return Err(format!("non-RSA key algo {algo}")); }
        if algo != 1 && algo != 3 { return Err(format!("key algo {algo} is not RSA encrypt-or-sign")); }
        let mut off = 6;
        let (nbits, n) = read_mpi(body, &mut off)?;
        let (_ebits, e) = read_mpi(body, &mut off)?;
        if n.is_empty() || e.is_empty() || nbits < 1024 { return Err("implausible RSA modulus".into()); }
        Ok((body[..off].to_vec(), n, e))
    }
    fn read_mpi(b: &[u8], off: &mut usize) -> Result<(usize, Vec<u8>), String> {
        if *off + 2 > b.len() { return Err("truncated MPI".into()); }
        let bits = u16::from_be_bytes([b[*off], b[*off + 1]]) as usize;
        let n = (bits + 7) / 8;
        if *off + 2 + n > b.len() { return Err("MPI overruns".into()); }
        let v = b[*off + 2..*off + 2 + n].to_vec();
        *off += 2 + n;
        Ok((bits, v))
    }
 
    // Key-fork subpacket (RFC 4880 §5.2.3.29): FIRST signature packet after
    // this key packet; type 27 carries the key flags byte (0x04/0x08 = may
    // encrypt). Byte-exact port of zkRsaPgpKeyFlags (zk-rsa.html ~646).
    fn key_flags(pks: &[Packet], idx: usize) -> Option<u8> {
        for p in pks.iter().skip(idx + 1) {
            match p.tag {
                5 | 6 | 7 | 14 => return None,
                2 => {
                    let b = &p.body;
                    if b[0] != 4 || b.len() < 6 { return None; }
                    let hlen = u16::from_be_bytes([b[4], b[5]]) as usize;
                    let mut pt = 6usize;
                    let end = 6 + hlen;
                    while pt < end && pt < b.len() {
                        let sl = if b[pt] < 192 { b[pt] as usize }
                            else if b[pt] < 254 { (((b[pt] as usize) - 192) << 8) + b[pt + 1] as usize + 192 }
                            else { u32::from_be_bytes([b[pt+1], b[pt+2], b[pt+3], b[pt+4]]) as usize };
                        if sl < 1 || pt + sl > b.len() { return None; }
                        if b[pt] == 27 && sl >= 2 { return Some(b[pt + 1]); }
                        pt += sl;
                    }
                    return None;
                }
                _ => {}
            }
        }
        None
    }
 
    pub struct ParsedRecipient { pub fingerprint: String, target: Target }
 
    /// Parse an armored OpenPGP public key and pick the ENCRYPTION target —
    /// identical selection rule to zk-rsa: RSA subkey with encrypt flags
    /// (0x04|0x08) → else LAST RSA subkey (OpenPGP [E]-subkey convention) →
    /// else the primary. Returns primary fingerprint (40 upper hex) + target
    /// (fingerprint, keyid, n, e).
    pub fn parse_public_key(armor: &str) -> Result<ParsedRecipient, String> {
        let bin = dearmor(armor)?;
        let pks = parse_packets(&bin)?;
        let mut keys: Vec<(usize, Vec<u8>, Vec<u8>, Vec<u8>, bool, Option<u8>)> = Vec::new(); // idx, material, n, e, is_primary, flags
        for (i, p) in pks.iter().enumerate() {
            match p.tag {
                6 | 14 => {
                    let (mat, n, e) = rsa_key(&p.body)?;
                    let is_primary = p.tag == 6;
                    let flags = if is_primary { None } else { key_flags(&pks, i) };
                    keys.push((i, mat, n, e, is_primary, flags));
                }
                _ => {}
            }
        }
        if keys.is_empty() { return Err("no RSA key packet found".into()); }
        let first = &keys[0];
        if !first.4 { return Err("first key packet is not a primary key".into()); }
        let fp = hex::encode(openssl::sha::sha1(&{
            let mut v = vec![0x99u8, (first.1.len() >> 8) as u8, first.1.len() as u8];
            v.extend_from_slice(&first.1);
            v
        })).to_uppercase();
        let pick = {
            let subs: Vec<&_> = keys.iter().filter(|k| !k.4).collect();
            if !subs.is_empty() {
                let flagged: Vec<&_> = subs.iter()
                    .filter(|s| matches!(s.5, Some(f) if f & 0x0c != 0)).collect();
                if !flagged.is_empty() { *flagged[flagged.len() - 1] } else { *subs[subs.len() - 1] }
            } else { first }
        };
        let keyid = fp[24..40].to_string(); // primary-keyid convention fallback (JS parity)
        let target_fp = {
            let mut v = vec![0x99u8, (pick.1.len() >> 8) as u8, pick.1.len() as u8];
            v.extend_from_slice(&pick.1);
            hex::encode(openssl::sha::sha1(&v)).to_uppercase()
        };
        Ok(ParsedRecipient {
            fingerprint: fp,
            target: Target { fingerprint: target_fp, keyid: target_fp[24..40].to_string(), n: pick.2.clone(), e: pick.3.clone() },
        })
    }
 
    /// Full message: PKESK v3 (target keyid, algo 1, PKCS#1 v1.5 of
    /// [0x09 ‖ sk ‖ checksum]) + SEIPD v1 (zero-IV AES-256-CFB-128 with NO
    /// §13.9 resync, prefix+literal+MDC) + standard armor. `plaintext` is the
    /// full envelope JSON.
    pub fn encrypt_message(armor_pub: &str, plaintext: &[u8]) -> Result<String, String> {
        let parsed = parse_public_key(armor_pub)?;
        let rsa = openssl::rsa::Rsa::public_public_components(
                openssl::bn::BigNum::from_slice(&parsed.target.n),
                openssl::bn::BigNum::from_slice(&parsed.target.e))
            .map_err(|e| format!("RSA public key: {e}"))?;
        let mut sk = [0u8; 32];
        openssl::rand::rand_bytes(&mut sk).map_err(|e| format!("rng: {e}"))?;
        let mut payload = Vec::with_capacity(35);
        payload.push(PGP_SYM_AES256);
        payload.extend_from_slice(&sk);
        let sum: u16 = sk.iter().fold(0u16, |a, &b| a.wrapping_add(b as u16));
        payload.extend_from_slice(&sum.to_be_bytes());
        let wrapped = rsa.public_encrypt(&payload, openssl::rsa::Padding::PKCS1)
            .map_err(|e| format!("PKCS#1 v1.5 wrap: {e}"))?;
 
        let mut pkesk = vec![0xc0u8 | 1, 3];          // new-format tag 1
        let kid = hex::decode(&parsed.target.keyid).map_err(|e| e.to_string())?;
        pkesk.extend_from_slice(&kid);
        pkesk.push(1);                                  // algo 1 = RSA (encrypt-or-sign)
        pkesk.extend_from_slice(&{
            let bits = rsa.size() * 8;
            let mut v = ((bits >> 8) as u16).to_be_bytes().to_vec();
            v.extend_from_slice(&wrapped);
            v
        });
 
        let mut literal = vec![0xc0u8 | 11, (6 + plaintext.len()) as u8 /*placeholder*/];
        // proper 2-byte length (6..8383 → two-octet form):
        let ll = 6 + plaintext.len();
        literal = if ll < 192 {
            vec![0xc0u8 | 11, ll as u8]
        } else {
            let v = ll - 192;
            vec![0xc0u8 | 11, ((v >> 8) + 192) as u8, (v & 0xff) as u8]
        };
        literal.extend_from_slice(&[0x62, 0x00, 0, 0, 0, 0]); // 'b', 0-name, mtime 0
        literal.extend_from_slice(plaintext);
 
        let mut prefix = [0u8; 18];
        openssl::rand::rand_bytes(&mut prefix[..16]).map_err(|e| format!("rng: {e}"))?;
        prefix[16] = prefix[14];
        prefix[17] = prefix[15];
        let mut pre_mdc: Vec<u8> = Vec::with_capacity(18 + literal.len() + 2);
        pre_mdc.extend_from_slice(&prefix);
        pre_mdc.extend_from_slice(&literal);
        pre_mdc.extend_from_slice(&[0xd3, 0x14]);        // MDC packet header (tag 19, len 20)
        let digest = openssl::sha::sha1(&pre_mdc);
        pre_mdc.extend_from_slice(&digest);
        let ct = openssl::symm::encrypt(openssl::symm::Cipher::aes_256_cfb128(), &sk, None, &pre_mdc)
            .map_err(|e| format!("CFB: {e}"))?;         // zero IV = no IV ⇒ zero IV ✓
 
        let mut seipd_body = vec![1u8];                  // SEIPD v1
        seipd_body.extend_from_slice(&ct);
        let mut seipd = vec![0xc0u8 | 18];
        seipd.extend_from_slice(&{
            let l = seipd_body.len();
            if l < 192 { vec![l as u8] }
            else { let v = l - 192; vec![((v >> 8) + 192) as u8, (v & 0xff) as u8] }
        });
        seipd.extend_from_slice(&seipd_body);
 
        let mut out = pkesk.clone();
        out.extend_from_slice(&seipd);
        Ok(armor_message(&out))
    }
 
    fn armor_message(bytes: &[u8]) -> String {
        use base64::Engine as _;
        let b64 = base64::engine::general_purpose::STANDARD.encode(bytes);
        let wrapped: String = b64.as_bytes().chunks(64)
            .map(|c| String::from_utf8(c.to_vec()).unwrap() + "\n").collect();
        let c = crc24(bytes);
        let crc = base64::engine::general_purpose::STANDARD.encode([(c >> 16) as u8, (c >> 8) as u8, c as u8]);
        format!("-----BEGIN PGP MESSAGE-----\n\n{wrapped}={crc}\n-----END PGP MESSAGE-----\n")
    }
 
    /// The second file: ciphertext + visible hashes. Envelope JSON goes in
    /// ENCRYPTED; the attested hashes sit in CLEAR (user request).
    pub fn build_sealed_report(
        armor_pub: &str,
        report_json: &str,
        raw_hash: &str,           // paypal_user_info_raw_hash (attested identity hash)
        id_hash: &str,            // SHA-256(user_id) — the literal "hash of the Paypal id"
    ) -> Result<String, String> {
        let parsed = parse_public_key(armor_pub)?;
        let ciphertext = encrypt_message(armor_pub, report_json.as_bytes())?;
        let sealed = serde_json::json!({
            "type": "pav-pgp-sealed-attestation/v1",
            "note": "CIPHERTEXT: the attestation envelope encrypted to recipient_fp. Only the holder of that PGP key's private half can decrypt (`gpg --decrypt`). The two hashes below are CLEAR by design so anyone can match this file to the plain attestation without reading its contents.",
            "recipient_fingerprint": parsed.fingerprint,
            "paypal_user_info_raw_hash": raw_hash,
            "paypal_id_sha256": id_hash,
            "recipient": "Bundeskriminalamt (BKA) — pgp-key4k@bka.bund.de",
            "ciphertext": ciphertext,
        });
        serde_json::to_string_pretty(&sealed).map_err(|e| e.to_string())
    }

// js
// v197 — fingerprint(s) of an armored PGP public key: SHA-1(0x99‖len‖material)
// over each v4 key packet (primary first), lowercase hex. Mirrors the
// enclave's pgp_escrow::parse_public_key so signer and auditor agree.
async function pgpPublicFingerprints(armor) {
    const m = /-----BEGIN PGP PUBLIC KEY BLOCK-----([\s\S]*?)-----END PGP PUBLIC KEY BLOCK-----/.exec(String(armor || ''));
    if (!m) throw new Error('not a PGP PUBLIC KEY BLOCK');
    const b64 = (m[1].split(/\n=/)[0].replace(/^[\s\S]*?\n\n/, '') || '').replace(/\s+/g, '');
    const bin = Uint8Array.from(atob(b64.replace(/=+$/, '')), c => c.charCodeAt(0));
    const out = []; let i = 0;
    while (i < bin.length) {
        const ctb = bin[i++]; let tag, len;
        if (ctb & 0x40) { tag = ctb & 0x3f;
            if (bin[i] < 192) { len = bin[i]; i += 1; }
            else if (bin[i] < 224) { len = (((bin[i] - 192) << 8) + bin[i + 1] + 192); i += 2; }
            else if (bin[i] === 255) { len = ((bin[i+1]<<24)>>>0) + (bin[i+2]<<16) + (bin[i+3]<<8) + bin[i+4]; i += 5; }
            else throw new Error('partial-body lengths unsupported');
        } else { tag = (ctb >> 2) & 0xf;
            if ((ctb & 3) === 0) { len = bin[i]; i += 1; }
            else if ((ctb & 3) === 1) { len = (bin[i] << 8) | bin[i + 1]; i += 2; }
            else if ((ctb & 3) === 2) { len = ((bin[i]<<24)>>>0) + (bin[i+1]<<16) + (bin[i+2]<<8) + bin[i+3]; i += 4; }
            else len = bin.length - i;
        }
        const body = bin.subarray(i, i + len); i += len;
        if (tag === 6 || tag === 14) {
            const pre = new Uint8Array(body.length + 3);
            pre[0] = 0x99; pre[1] = body.length >> 8; pre[2] = body.length & 0xff;
            pre.set(body, 3);
            const d = await crypto.subtle.digest('SHA-1', pre);
            out.push(Array.from(new Uint8Array(d)).map(b => b.toString(16).padStart(2, '0')).join('').toUpperCase());
        }
    }
    if (!out.length) throw new Error('no OpenPGP key packets found');
    return out;
}
